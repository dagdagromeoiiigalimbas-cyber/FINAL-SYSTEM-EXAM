My final CC102 Performance Task: athletes jump analyzer.

import java.util.Scanner;

public class LongJumpAnalyzer {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        final int SIZE = 12;
        double[] jumps = new double[SIZE];
        for (int i = 0; i < SIZE; i++) {
            double value;
            while (true) {
                System.out.print("Enter distance for athlete " + (i + 1) + " (meters): ");
                if (sc.hasNextDouble()) {
                    value = sc.nextDouble();
                    if (value >= 0) break;
                    System.out.println("Invalid: value must be nonnegative.");
                } else {
                    System.out.println("Invalid: please enter a number.");
                    sc.next(); 
                }
            }
            jumps[i] = value;
        }
        double sum = 0;
        double longest = jumps[0], shortest = jumps[0];
        int longestIdx = 0, shortestIdx = 0;
        int count = 0;

        for (int i = 0; i < SIZE; i++) {
            sum += jumps[i];
            if (jumps[i] > longest) {
                longest = jumps[i];
                longestIdx = i;
            }
            if (jumps[i] < shortest) {
                shortest = jumps[i];
                shortestIdx = i;
            }
            if (jumps[i] >= 4.0) count++;
        }
        double average = sum / SIZE;
        average = Math.round(average * 100) / 100.0; 
        for (int i = 0; i < SIZE - 1; i++) {
            int maxIdx = i;
            for (int j = i + 1; j < SIZE; j++) {
                if (jumps[j] > jumps[maxIdx]) maxIdx = j;
            }
            double temp = jumps[i];
            jumps[i] = jumps[maxIdx];
            jumps[maxIdx] = temp;
        }
        System.out.println();
        System.out.println("--- Results ---");
        System.out.println("Average distance: " + average + " m");
        System.out.println("Longest jump: " + longest + " m (index " + longestIdx + " in original input)");
        System.out.println("Shortest jump: " + shortest + " m (index " + shortestIdx + " in original input)");
        System.out.println("Athletes jumping at least 4 meters: " + count);

        System.out.println();
        System.out.println("Distances in descending order:");
        for (int i = 0; i < SIZE; i++) {
            System.out.println("[" + i + "] " + jumps[i] + " m");
        }
    }
}
